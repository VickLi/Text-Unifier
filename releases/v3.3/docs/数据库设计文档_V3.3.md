## 前言

V3.3 在 V3.2 基础上新增了**双向切换按钮状态**（全角/繁简转换状态）、**Minimap 色条数据**、**关键词搜索位置**和**段落修改跟踪**。IndexedDB 层新增 `is_fullwidth_converted` 和 `is_traditional_converted` 两个用户偏好，用于会话恢复。

### 3. 派生状态（V3.3 新增/变更）

```typescript
// === 段落状态判定（ParagraphIndicator 用） ===
function getParagraphState(
    para: PreviewParagraph,
    checkedMap: Map<string, boolean>,
    modifiedIds: Set<string>
): ParagraphState {
    if (checkedMap.get(para.id) === false) return 'deleted';   // 红色
    if (modifiedIds.has(para.id)) return 'modified';            // 橙色
    return 'normal';                                             // 绿色
}

// === Minimap 数据派生 ===
function computeMinimapItems(
    paragraphs: PreviewParagraph[],
    checkedMap: Map<string, boolean>,
    modifiedIds: Set<string>
): MinimapItem[] {
    return paragraphs.map(p => ({
        color: getParagraphState(p, checkedMap, modifiedIds) === 'deleted' ? 'red'
             : getParagraphState(p, checkedMap, modifiedIds) === 'modified' ? 'orange'
             : 'green',
        tooltip: p.text.slice(0, 20),
    }));
}

// === 搜索是否有匹配 ===
const hasKeywordMatch = (state: TextUnifierStore): boolean => {
    if (!state.cleanOptions.filterKeywords) return false;
    const kws = state.cleanOptions.filterKeywords.split(',').map(k => k.trim()).filter(Boolean);
    if (kws.length === 0) return false;
    const text = state.previewParagraphs
        .filter(p => state.paragraphCheckedMap.get(p.id) !== false)
        .map(p => p.text)
        .join('\n');
    return kws.some(kw => text.includes(kw));
};

// === 撤回栈深度（显示用） ===
const undoDepthLabel = (state: TextUnifierStore): string => {
    if (state.undoStack.length === 0) return '';
    return `${state.undoPointer + 1}/${state.undoStack.length}`;
};
```

### 4. 核心 Action 实现逻辑（V3.3 新增/变更）

#### 4.1 `toggleFullWidth` — 全角↔半角双向切换

```typescript
toggleFullWidth: () =>
    set((state) => {
        const newConverted = !state.isFullWidthConverted;
        const checkedParas = state.previewParagraphs.filter(
            p => state.paragraphCheckedMap.get(p.id) !== false
        );
        const text = checkedParas.map(p => p.text).join('\n\n');

        let resultText: string;
        if (newConverted) {
            resultText = toHalfWidth(text);   // TS 纯函数
        } else {
            resultText = toFullWidth(text);   // 反向恢复
        }

        const formattedTexts = resultText.split('\n\n');
        const newParagraphs = state.previewParagraphs.map(p => {
            const idx = checkedParas.findIndex(cp => cp.id === p.id);
            if (idx >= 0 && idx < formattedTexts.length && p.text !== formattedTexts[idx]) {
                return { ...p, text: formattedTexts[idx], content_hash: computeHash(formattedTexts[idx]) };
            }
            return p;
        });

        // 标记被修改的段落
        const modifiedIds = new Set(state.modifiedParagraphIds);
        newParagraphs.forEach(np => {
            const orig = state.previewParagraphs.find(p => p.id === np.id);
            if (orig && orig.text !== np.text) modifiedIds.add(np.id);
        });

        return {
            isFullWidthConverted: newConverted,
            previewParagraphs: newParagraphs,
            modifiedParagraphIds: modifiedIds,
            minimapItems: computeMinimapItems(newParagraphs, state.paragraphCheckedMap, modifiedIds),
        };
    }),
```

#### 4.2 `searchNextKeyword` — 搜索下一个

```typescript
searchNextKeyword: () =>
    set((state) => {
        const kws = state.cleanOptions.filterKeywords
            .split(',').map(k => k.trim()).filter(Boolean);
        if (kws.length === 0) return state;

        const text = state.previewParagraphs
            .filter(p => state.paragraphCheckedMap.get(p.id) !== false)
            .map(p => p.text)
            .join('\n');
        const lines = text.split('\n');

        // 从当前位置+1开始搜索
        let foundIdx: number | null = null;
        for (let i = state.keywordSearchIndex + 1; i < lines.length; i++) {
            if (kws.some(kw => kw && lines[i].includes(kw))) {
                foundIdx = i;
                break;
            }
        }
        // wrap-around: 从头搜索
        if (foundIdx === null) {
            for (let i = 0; i <= state.keywordSearchIndex; i++) {
                if (kws.some(kw => kw && lines[i].includes(kw))) {
                    foundIdx = i;
                    break;
                }
            }
        }

        if (foundIdx === null) {
            // Toast "未找到匹配行"
            return { keywordSearchIndex: 0, highlightedLineIndex: null };
        }

        return {
            keywordSearchIndex: foundIdx,
            highlightedLineIndex: foundIdx,
        };
        // 高亮消退由 useEffect 处理：setTimeout 3s → highlightedLineIndex = null
    }),
```

#### 4.3 `deleteAllKeywordMatches` — 一键删除

```typescript
deleteAllKeywordMatches: () =>
    set((state) => {
        const kws = state.cleanOptions.filterKeywords
            .split(',').map(k => k.trim()).filter(Boolean);
        if (kws.length === 0) return state;

        const newParagraphs = state.previewParagraphs.map(p => {
            if (state.paragraphCheckedMap.get(p.id) === false) return p;
            const filtered = filterLines(p.text, kws, 0);
            if (filtered !== p.text) {
                return { ...p, text: filtered, content_hash: computeHash(filtered) };
            }
            return p;
        });

        return {
            previewParagraphs: newParagraphs,
            keywordSearchIndex: 0,
            highlightedLineIndex: null,
        };
    }),
```

#### 4.4 `removeFile` — 删除文件增强版（RQ-03）

```typescript
removeFile: (path: string) =>
    set((state) => {
        const removedName = path.split(/[/\\]/).pop() || '';

        // 更新文件列表
        const newSortedList = state.sortedFileList.filter(f => f.path !== path);
        const newFileList = state.fileList.filter(f => f.path !== path);

        // 更新重复组
        const newGroups = state.duplicateGroups
            .map(group => {
                const newSources = group.sources.filter(s => s.file_name !== removedName);
                const uniqueFiles = new Set(newSources.map(s => s.file_name));
                return { ...group, sources: newSources, occurrence_count: uniqueFiles.size };
            })
            .filter(g => g.occurrence_count > 1);

        return {
            fileList: newFileList,
            sortedFileList: newSortedList,
            duplicateGroups: newGroups,
            modifiedParagraphIds: new Set(),
            clearUndoStack: true,  // 结构调整后撤回栈失效
        };
    }),
```

#### 4.5 `revertBeforeExport` — 导出前还原

```typescript
revertBeforeExport: () =>
    set((state) => {
        if (state.undoPointer <= 0) return state;  // 无快照可还原
        // 等价于 undo()，但操作原因标记为"导出前还原"
        const newPointer = state.undoPointer - 1;
        const snap = state.undoStack[newPointer];
        return {
            undoPointer: newPointer,
            previewParagraphs: structuredClone(snap.paragraphs),
            paragraphCheckedMap: new Map(Object.entries(snap.checkedMap)),
        };
    }),
```
